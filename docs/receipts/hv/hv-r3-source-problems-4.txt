710: function Stop-ReplChildTree {
711:     <#
712:     .SYNOPSIS
713:         Close stdin, kill the child process tree, and await owned pipe closure.
714:     .DESCRIPTION
715:         Process.WaitForExit waits only for the root process after Kill(true). Redirected stdout
716:         and stderr remain open while a killed descendant is still exiting, so require those
717:         drain tasks to complete inside the same cleanup reserve before confirming termination.
718:     #>
719:     param(
720:         $Process,
721:         $StdoutTask,
722:         $StderrTask,
723:         [int]$WaitMs = 500
724:     )
725: 
726:     if ($null -eq $Process) { return $true }
727:     try {
728:         $Process.StandardInput.BaseStream.Dispose()
729:     } catch {
730:     }
731:     try {
732:         $Process.StandardInput.Dispose()
733:     } catch {
734:     }
735:     try {
736:         if (-not $Process.HasExited) {
737:             $Process.Kill($true)
738:         }
739:     } catch {
740:     }
741: 
742:     $cleanup = [System.Diagnostics.Stopwatch]::StartNew()
743:     do {
744:         $rootExited = $false
745:         try { $rootExited = $Process.HasExited } catch { $rootExited = $true }
746:         $stdoutClosed = -not $StdoutTask -or $StdoutTask.IsCompleted
747:         $stderrClosed = -not $StderrTask -or $StderrTask.IsCompleted
748:         if ($rootExited -and $stdoutClosed -and $stderrClosed) {
749:             return $true
750:         }
751:         Start-Sleep -Milliseconds 10
752:     } while ($cleanup.ElapsedMilliseconds -lt $WaitMs)
753: 
754:     return $false
755: }
