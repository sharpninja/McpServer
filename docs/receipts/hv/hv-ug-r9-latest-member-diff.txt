+            var separator = group.IndexOf(':', StringComparison.Ordinal);
+            if (separator < 0)

+                throw new InvalidOperationException($"Test group '{group}' names test project '{project}', which the solution does not contain.");
+
+            var names = group[(separator + 1)..].Split(',').Select(name => name.Trim()).ToArray();

+                throw new InvalidOperationException($"Test group '{group}' has an empty class name.");

+                }
+            }
+
+            if (!classes.TryGetValue(project, out var selected))

+                selected = [];
+                classes[project] = selected;
+                order.Add(project);
+            }

+            foreach (var name in names.Where(name => !selected.Contains(name, StringComparer.Ordinal)))
+                selected.Add(name);
+        }

+    }

+        {

+            // A result's display name normally starts with its class; a TestMethod definition names the class itself,
+            // which still holds when a display name is customized.
+            testNames = document.Descendants()
+                .Where(element => element.Name.LocalName == "UnitTestResult")
+                .Select(element => (string?)element.Attribute("testName") ?? string.Empty)

+                    .Where(element => element.Name.LocalName == "TestMethod")

+            ? null
+            : $"Test group project '{project}' ran no case of class {string.Join(", ", missing)}.";
+    }

+namespace QBrainAi.Support.Mcp.Requirements;

+/// <summary>
+/// FR-MCP-040 ac-1: reads the creation and update times stored with a requirement or traceability link, so a read reports
+/// the stored time and never the time of the read. Stored values are round-trip text written in UTC; a blank or
+/// unreadable value reads as null.
+/// </summary>
+/// <remarks>
+/// <list type="table">
+/// <listheader><term>Date</term><description>Requirements; AC; change and reason</description></listheader>
+/// <item><term>2026-11-02</term><description>FR-MCP-040 ac-1: added Read and Earliest, because requirement reads reported the time of the read instead of the stored times (operator 2026-11-02: "This HAS to be fixed.").</description></item>
+/// </list>
+/// </remarks>
+internal static class RequirementStoredTimestamp
+{
+    /// <summary>Reads one stored time, keeping its stored offset.</summary>
+    /// <param name="stored">The stored text.</param>
+    /// <returns>The stored time, or null when nothing readable is stored.</returns>
+    internal static DateTimeOffset? Read(string? stored) =>
+        !string.IsNullOrWhiteSpace(stored)
+        && DateTimeOffset.TryParse(stored, CultureInfo.InvariantCulture, DateTimeStyles.AssumeUniversal, out var value)
+            ? value
+            : null;
+
+    /// <summary>Reads the earliest readable stored time, for a mapping that aggregates several links.</summary>
+    /// <param name="stored">The links' stored creation texts.</param>
+    /// <returns>The earliest stored time, or null when none is readable.</returns>
+    internal static DateTimeOffset? Earliest(IEnumerable<string?> stored)
+    {
+        ArgumentNullException.ThrowIfNull(stored);
+        DateTimeOffset? earliest = null;
+        foreach (var value in stored.Select(Read))
+        {
+            if (value is not null && (earliest is null || value.Value < earliest.Value))
+                earliest = value;
+        }
+
+        return earliest;
+    }
