+namespace NukeBuild.Tests;
+
+/// <summary>
+/// TR-MCP-NUKE-019 (AC-1): Contract cases and assertions for <see cref="IIntegrationLaneFilter"/>. The expected filters
+/// come from the acceptance criterion; a small evaluator of the trait grammar the lane uses (terms joined by '&amp;',
+/// Name=Value requires the trait, Name!=Value is met by any other value or by no trait) checks what each filter
+/// selects for an untagged case, an own-host case, an other-host case and a non-integration case. The same
+/// assertions run against the scripted filter (BDPv4 mock gate), wrong filters (negative controls), and the
+/// production filter.
+/// </summary>
+internal static class IntegrationLaneFilterContract
+{
+    /// <summary>Every host name, in the order the contract checks them.</summary>
+    internal static readonly string[] HostNames = ["windows", "linux"];
+
+    /// <summary>Parses a contract host name.</summary>
+    /// <param name="hostName">windows or linux.</param>
+    internal static IntegrationLaneHost Parse(string hostName) => hostName switch
+    {
+        "windows" => IntegrationLaneHost.Windows,
+        "linux" => IntegrationLaneHost.Linux,
+        _ => throw new ArgumentOutOfRangeException(nameof(hostName), hostName, "Unknown contract host."),
+    };
+
+    /// <summary>The filter TR-MCP-NUKE-019 AC-1 requires for <paramref name="host"/>.</summary>
+    /// <param name="host">Host the lane runs on.</param>
+    internal static string ExpectedFilter(IntegrationLaneHost host) => host switch
+    {
+        IntegrationLaneHost.Windows => "Category=Integration&Platform!=Linux",
+        IntegrationLaneHost.Linux => "Category=Integration&Platform!=Windows",
+        _ => throw new ArgumentOutOfRangeException(nameof(host), host, "Unknown host."),
+    };
+
+    /// <summary>Asserts that <paramref name="filter"/> meets the contract for the named host.</summary>
+    /// <param name="hostName">windows or linux.</param>
+    /// <param name="filter">Filter under test.</param>
+    internal static void AssertHost(string hostName, IIntegrationLaneFilter filter)
+    {
+        var host = Parse(hostName);
+        var actual = filter.Compose(host);
+
+        Assert.Equal(ExpectedFilter(host), actual);
+        AssertSelection(host, actual);
+    }
+
+    /// <summary>
+    /// Asserts what <paramref name="filter"/> selects on <paramref name="host"/>: untagged and own-host integration
+    /// cases run; other-host and non-integration cases do not. Independent of the exact filter text, so the negative
+    /// controls prove each of these assertions rejects a wrong filter on its own.
+    /// </summary>
+    /// <param name="host">Host the lane runs on.</param>
+    /// <param name="filter">Filter text under test.</param>
+    internal static void AssertSelection(IntegrationLaneHost host, string filter)
+    {
+        var own = host == IntegrationLaneHost.Windows ? "Windows" : "Linux";
+        var other = host == IntegrationLaneHost.Windows ? "Linux" : "Windows";
+
+        Assert.True(Selects(filter, category: "Integration", platform: null), "An untagged integration case must run on every host.");
+        Assert.True(Selects(filter, category: "Integration", platform: own), $"A Platform={own} integration case must run on its own host.");
+        Assert.False(Selects(filter, category: "Integration", platform: other), $"A Platform={other} integration case must not run on a {own} host.");
+        Assert.False(Selects(filter, category: null, platform: null), "A case without Category=Integration must not run in the integration lane.");
+    }
+
+    /// <summary>Evaluates the '&amp;'-joined Name=Value and Name!=Value terms against one case's traits.</summary>
+    /// <param name="filter">Filter text.</param>
+    /// <param name="category">The case's Category trait, or null.</param>
+    /// <param name="platform">The case's Platform trait, or null.</param>
+    internal static bool Selects(string filter, string? category, string? platform)
+    {
+        var traits = new Dictionary<string, string?>(StringComparer.Ordinal)
+        {
+            ["Category"] = category,
+            ["Platform"] = platform,
+        };
+        foreach (var term in filter.Split('&', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries))
+        {
+            var negated = term.Contains("!=", StringComparison.Ordinal);
+            var parts = term.Split(negated ? "!=" : "=", 2, StringSplitOptions.TrimEntries);
+            if (parts.Length != 2 || !traits.TryGetValue(parts[0], out var value))
+            {
+                throw new FormatException($"Unsupported filter term '{term}'.");
+            }
+
+            var equal = string.Equals(value, parts[1], StringComparison.Ordinal);
+            if (negated ? equal : !equal)
+            {
+                return false;
+            }
+        }
+
+        return true;
+    }
+}

+/// <summary>
+/// TR-MCP-NUKE-019 (AC-5): contract scenarios and assertions for <see cref="IIntegrationLanePrecondition"/>. The
+/// expected outcomes come from the criterion: on Windows the lane runs only from an interactive desktop session
+/// and otherwise fails before any test, naming the missing session; on Linux no session requirement applies. The

+/// </summary>
+internal static class IntegrationLanePreconditionContract
+{
+    /// <summary>Every scenario name, in the order the contract checks them.</summary>

+    [
+        "windows-interactive",
+        "windows-noninteractive",
+        "linux-interactive",
+        "linux-noninteractive",
+    ];
+
+    /// <summary>Asserts that <paramref name="precondition"/> meets the named scenario.</summary>
+    /// <param name="scenario">Scenario name from <see cref="ScenarioNames"/>.</param>
+    /// <param name="precondition">Precondition under test.</param>
+    internal static void AssertScenario(string scenario, IIntegrationLanePrecondition precondition)
+    {
+        var (host, interactive, mayRun) = scenario switch
+        {
+            "windows-interactive" => (IntegrationLaneHost.Windows, true, true),
+            "windows-noninteractive" => (IntegrationLaneHost.Windows, false, false),
+            "linux-interactive" => (IntegrationLaneHost.Linux, true, true),
+            "linux-noninteractive" => (IntegrationLaneHost.Linux, false, true),
+            _ => throw new ArgumentOutOfRangeException(nameof(scenario), scenario, "Unknown contract scenario."),
+        };
+
+        var readiness = precondition.Evaluate(host, interactive);
+
+        Assert.Equal(mayRun, readiness.CanRun);
+        if (mayRun)
+        {
+            Assert.Null(readiness.FailureMessage);
+            return;
+        }
+
+        Assert.False(string.IsNullOrWhiteSpace(readiness.FailureMessage), "A lane that cannot run must say why instead of stopping silently.");
+        Assert.Contains("interactive desktop session", readiness.FailureMessage, StringComparison.Ordinal);
+        Assert.Contains("MigrationIntegrationTests", readiness.FailureMessage, StringComparison.Ordinal);
+    }

+using Microsoft.Data.Sqlite;
+
+namespace QBrainAi.Support.Mcp.Tests.Storage;

+/// <summary>
+/// TR-MCP-DB-001: raw-SQL workspace-key assertions for SQLite data-migration tests. A data migration copies the parent
+/// workspace key into every backfilled child row (the child foreign key to Workspaces). EF query parameters compared
+/// with workspace keys are canonicalized by WorkspaceKeyConversion, so a legacy key (for example a backslash Windows
+/// spelling) is only observable through SQL that bypasses EF.
+/// </summary>
+internal static class MigrationKeyAssertions
+{
+    /// <summary>
+    /// Asserts that the only non-global workspace row has exactly <paramref name="expectedKey"/> as its WorkspaceId,
+    /// and that <paramref name="childTable"/> has rows whose WorkspaceId values are all exactly that key.
+    /// </summary>
+    /// <param name="connection">Open connection to the migrated database.</param>
+    /// <param name="childTable">Backfilled child table that carries a WorkspaceId column.</param>
+    /// <param name="expectedKey">The workspace key seeded before the migration, compared ordinally.</param>
+    internal static void AssertChildRowsCarryWorkspaceKey(SqliteConnection connection, string childTable, string expectedKey)
+    {
+        ArgumentNullException.ThrowIfNull(connection);
+        ArgumentException.ThrowIfNullOrWhiteSpace(childTable);
+        ArgumentNullException.ThrowIfNull(expectedKey);
+
+        var parentKeys = ReadStrings(connection, "SELECT \"WorkspaceId\" FROM \"Workspaces\" WHERE \"WorkspaceId\" <> '' ORDER BY 1;");
+        Assert.Equal([expectedKey], parentKeys);
+
+        var childKeys = ReadStrings(connection, $"SELECT DISTINCT \"WorkspaceId\" FROM \"{childTable}\" ORDER BY 1;");
+        Assert.Equal([expectedKey], childKeys);
+    }
+
+    private static List<string> ReadStrings(SqliteConnection connection, string sql)
+    {
+        using var command = connection.CreateCommand();
+        command.CommandText = sql;
+        using var reader = command.ExecuteReader();
+        var values = new List<string>();
+        while (reader.Read())
+        {
+            values.Add(reader.GetString(0));
+        }
+
+        return values;
+    }

+
+    /// <summary>On a Windows host, Windows-platform paths switch to backslashes.</summary>
+    [Theory]
+    [InlineData("C:/Users/kingd/repo", @"C:\Users\kingd\repo")]
+    [InlineData("C:/", @"C:\")]
+    [InlineData("//srv/share/x", @"\\srv\share\x")]
+    [InlineData(@"C:\already\native", @"C:\already\native")]
+    public void WindowsHost_WindowsPath_UsesBackslashes(string path, string expected)
+    {
+        AssertNative(expected, ToNativePath(path, WorkspacePathPlatform.Windows, SimulatedWorkspaceHosts.Windows()));
+    }
+
+    /// <summary>On a non-Windows host nothing is rewritten, whatever the platform tag.</summary>
+    [Theory]
+    [InlineData("C:/Users/kingd/repo", WorkspacePathPlatform.Windows)]
+    [InlineData("//srv/share/x", WorkspacePathPlatform.Windows)]
+    [InlineData("/home/x/repo", WorkspacePathPlatform.CaseSensitive)]
+    public void NonWindowsHost_IsUnchanged(string path, WorkspacePathPlatform platform)
+    {
+        AssertUnchanged(path, ToNativePath(path, platform, SimulatedWorkspaceHosts.Linux()));
+    }
+
+    /// <summary>Assertion shared by the non-Windows-host contract and its negative controls.</summary>
+    /// <param name="path">Path given to the native-spelling function.</param>
+    /// <param name="actual">Value the function returned.</param>
+    protected static void AssertUnchanged(string path, string actual) => Assert.Equal(path, actual);
+
+    /// <summary>Assertion shared by the Windows-host contract and its negative controls.</summary>
+    /// <param name="expected">Backslash spelling the requirement defines.</param>
+    /// <param name="actual">Value the function returned.</param>
+    protected static void AssertNative(string expected, string actual) => Assert.Equal(expected, actual);
